#!/bin/zsh
# usage: run.sh <name> <kind: unit|storage|both> then triples: file old new
set -u
W=/Users/mac/code/Natively/.claude/worktrees/margin-agpl-migration
S=/private/tmp/claude-501/-Users-mac-code-Natively/9d4dc2ab-eb42-43c2-ab59-04e8fddb2ec8/scratchpad/r7m
name=$1; kind=$2; shift 2
cd $W
files=()
while (( $# >= 3 )); do
  f=$1; old=$2; new=$3; shift 3
  cp $f $S/backup.$(basename $f)
  files+=$f
  if [[ $old == NEAR:* ]]; then python3 -I $S/mutate_near.py $f "${old#NEAR:}" "${new%%|||*}" "${new#*|||}"; else python3 -I $S/mutate.py $f "$old" "$new"; fi || { echo "MUTATE FAILED $f"; }
done
if [[ $kind == unit || $kind == both ]]; then
  node scripts/margin/isolation/run-node.mjs --test electron/services/__tests__/ScopeLocalFallback.test.mjs > $S/$name.unit.log 2>&1; echo "$name unit exit=$?"
fi
if [[ $kind == storage || $kind == both ]]; then
  node scripts/margin/isolation/run-node.mjs --allow-exec Electron --test --test-timeout=300000 tests/margin/meeting-storage.test.mjs > $S/$name.storage.log 2>&1; echo "$name storage exit=$?"
fi
for f in $files; do
  cp $S/backup.$(basename $f) $f
  cmp $S/backup.$(basename $f) $f && echo "restored $f"
done
